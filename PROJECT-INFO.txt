DATA MODEL
admins:
  user_id — Supabase Auth user UUID
  created_at — admin creation time

bots:
  id — UUID
  name — display title
  username — Telegram username, unique
  category — category name
  description — short details
  tags — text array
  image_url — public image URL
  image_path — storage object path
  status — active or maintenance
  featured — featured card
  is_public — public visibility
  clicks — click counter placeholder for future analytics
  created_by — admin UUID
  created_at / updated_at — timestamps

bot_categories:
  name, enabled, sort_order
